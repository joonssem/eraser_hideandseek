/* 기믹 1) 미끄럼 존 — 관성 유지, 입력 방향 전환 둔화(술래·지우개 모두) */
const Slip={vel:new THREE.Vector3(),last:new THREE.Vector3(),init:false};
function updateSlipZones(dt){
  if(G.screen!=="GAME"||G.role==="ghost"||G.spectating||dt<=0){ Slip.last.copy(G.pos); Slip.init=true; return; }
  if(!Slip.init){ Slip.last.copy(G.pos); Slip.init=true; return; }
  const vx=(G.pos.x-Slip.last.x)/dt, vz=(G.pos.z-Slip.last.z)/dt;
  let inZone=false;
  for(const s of CAFE_SLIP){ const dx=G.pos.x-s.x,dz=G.pos.z-s.z; if(dx*dx+dz*dz<s.r*s.r&&G.pos.y<1.2){inZone=true;break;} }
  if(inZone){
    Slip.vel.x=Slip.vel.x*0.985+(vx-Slip.vel.x)*0.20;   // 감쇠 0.985, 입력 가중치 20%
    Slip.vel.z=Slip.vel.z*0.985+(vz-Slip.vel.z)*0.20;
    moveHorizontalAxis("x",Slip.vel.x*dt*0.7);
    moveHorizontalAxis("z",Slip.vel.z*dt*0.7);
  }else{ Slip.vel.set(vx,0,vz); }
  Slip.last.copy(G.pos);
}
/* 사각 루프 경로 위 거리 d(둘레) → 좌표·진행방향 */
function beltPathPos(d,b){
  const Lx=b.xR-b.xL, Lz=b.zFar-b.zNear;
  d=((d%b.P)+b.P)%b.P;
  if(d<Lx) return {x:b.xL+d, z:b.zNear, dx:1, dz:0};   d-=Lx;
  if(d<Lz) return {x:b.xR, z:b.zNear+d, dx:0, dz:1};   d-=Lz;
  if(d<Lx) return {x:b.xR-d, z:b.zFar, dx:-1, dz:0};   d-=Lx;
  return {x:b.xL, z:b.zFar-d, dx:0, dz:-1};
}
/* 플레이어 위치를 사각 루프 경로에 투영 → 둘레거리 d와 경로선까지 거리 반환 */
function beltProject(px,pz,b){
  const Lx=b.xR-b.xL, Lz=b.zFar-b.zNear;
  const cx=clamp(px,b.xL,b.xR), cz=clamp(pz,b.zNear,b.zFar);
  const edges=[
    {ex:cx,  ez:b.zNear, d:(cx-b.xL)},          // 앞 변 A→B (+x)
    {ex:b.xR,ez:cz,      d:Lx+(cz-b.zNear)},     // 우 변 B→C (+z)
    {ex:cx,  ez:b.zFar,  d:Lx+Lz+(b.xR-cx)},     // 뒤 변 C→D (-x)
    {ex:b.xL,ez:cz,      d:2*Lx+Lz+(b.zFar-cz)}, // 좌 변 D→A (-z)
  ];
  let best=edges[0], bd=Infinity;
  for(const e of edges){ const ddx=px-e.ex, ddz=pz-e.ez, dist=ddx*ddx+ddz*ddz; if(dist<bd){bd=dist;best=e;} }
  return {d:best.d, dist:Math.sqrt(bd)};
}
/* 기믹 2) 순환 컨베이어 — 식판이 루프를 계속 돌고, 벨트 위 플레이어도 루프를 따라 코너까지 계속 실려 감 */

  if(!CAFE_BELT)return;
  const b=CAFE_BELT, t=(Net.now?Net.now():Date.now())/1000;
  // 식판 순환(항상)
  CAFE_TRAYS.forEach(tr=>{ const p=beltPathPos(b.speed*t+tr.d0,b);
    tr.mesh.position.set(p.x,b.top+0.35,p.z); tr.mesh.rotation.y=(p.dz!==0)?Math.PI/2:0; });
